#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Keypad.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <DHT.h>
#include <ArduinoJson.h> 
#include "RTClib.h" 

// ------------------ NETWORK CONFIGURATION ------------------
const char* ssid = "YOUR_WIFI_SSID"; 
const char* password = "YOUR_WIFI_PASSWORD";
const char* serverName = "https://smart-shelf-backend-lbmm.onrender.com/api/inventory";

// ------------------ HARDWARE PINS (Physical Specs) ------------------
#define DHTPIN 15
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

RTC_DS1307 rtc; 

#define RED_PIN 2
#define GREEN_PIN 0
#define BLUE_PIN 5
#define BUZZER_PIN 23

// I2C LCD (SDA:21, SCL:22)
LiquidCrystal_I2C lcd(0x27, 16, 2);

const byte ROWS = 4;
const byte COLS = 4;
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {13, 12, 14, 27};
byte colPins[COLS] = {26, 25, 33, 32};
Keypad customKeypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// ------------------ DATA STRUCTURES ------------------
struct Item {
  char name[9];
  int expiryDay;
  int expiryMonth;
  bool active = false;
};

const int MAX_ITEMS = 10;
Item inventory[MAX_ITEMS];
float currentTemp = 0.0;

enum SystemState {
  STATE_IDLE,
  STATE_SCROLL_MENU,
  STATE_CUSTOM_ENTRY,
  STATE_EXPIRY_ENTRY,
  STATE_VIEW_INVENTORY,
  STATE_VIEW_RECIPES
};
SystemState currentState = STATE_IDLE;

const char* predefinedItems[] = { "Milk", "Bread", "Eggs", "Cheese", "Butter", "Rice", "Flour", "Other" };
const int NUM_PREDEFINED = 8;
const char* recipes[] = {"Pancakes", "Omelette", "Sandwich"};
const int NUM_RECIPES = 3;

char itemNameInput[9];
int scrollIndex = 0;
int inventoryIndex = 0;
int recipeIndex = 0;

int charIndex = 0;
char lastKey = ' ';
unsigned long lastKeyTime = 0;
int multiTapIndex = 0;
const int multiTapTimeout = 1000;

char dateInput[5]; // Stores 4 digits (DDMM) + null terminator
int dateIdx = 0;

// ------------------ HARDWARE HELPER FUNCTIONS ------------------

void setStatusLED(int r, int g, int b) {
  analogWrite(RED_PIN, 255 - r);
  analogWrite(GREEN_PIN, 255 - g);
  analogWrite(BLUE_PIN, 255 - b);
}

void beepBuzzer(int times = 1) {
  for(int i = 0; i < times; i++) {
    tone(BUZZER_PIN, 1000); 
    delay(150);
    noTone(BUZZER_PIN);
    if(times > 1) delay(100);
  }
}

void showIdleScreen() {
  DateTime now = rtc.now(); 
  lcd.clear();
  lcd.noCursor();
  lcd.setCursor(0,0);
  lcd.print(now.day()); lcd.print("/"); lcd.print(now.month());
  lcd.setCursor(9,0);
  lcd.print(currentTemp, 1); lcd.print("C");
  lcd.setCursor(0,1);
  lcd.print("#Add B:Inv C:Rec");
  currentState = STATE_IDLE;
}

// ------------------ CLOUD SYNC LOGIC ------------------

void fetchInventory() {
  if(WiFi.status() == WL_CONNECTED) {
    lcd.clear(); lcd.print("Syncing Cloud...");
    HTTPClient http;
    http.begin(serverName);
    int httpCode = http.GET();
    if (httpCode == HTTP_CODE_OK) {
      String payload = http.getString();
      DynamicJsonDocument doc(4096);
      deserializeJson(doc, payload);
      for(int i=0; i<MAX_ITEMS; i++) inventory[i].active = false;
      JsonArray array = doc.as<JsonArray>();
      int i = 0;
      for (JsonObject obj : array) {
        if (i >= MAX_ITEMS) break;
        strncpy(inventory[i].name, obj["name"], 8);
        const char* date = obj["expiryDate"];
        if (date && strlen(date) >= 10) {
          inventory[i].expiryMonth = (date[5]-'0')*10 + (date[6]-'0');
          inventory[i].expiryDay = (date[8]-'0')*10 + (date[9]-'0');
        }
        inventory[i].active = true;
        i++;
      }
      lcd.clear(); lcd.print("Sync Complete!");
      beepBuzzer(2);
      delay(1500); 
    } else {
      lcd.clear(); lcd.print("Sync Error!");
      delay(1000);
    }
    http.end();
  }
}

void manageAndCheckExpiry() {
  float t = dht.readTemperature();
  if (!isnan(t)) currentTemp = t;
  DateTime now = rtc.now(); 
  bool hasExpired = false, hasWarning = false;
  for(int i = 0; i < MAX_ITEMS; i++) {
    if(inventory[i].active) {
      int dayOffset = (currentTemp > 32.0) ? 1 : 0;
      if(inventory[i].expiryMonth == now.month()) {
        int diff = inventory[i].expiryDay - (now.day() + dayOffset);
        if (diff <= 0) hasExpired = true;
        else if (diff <= 2) hasWarning = true;
      } else if (inventory[i].expiryMonth < now.month()) hasExpired = true;
    }
  }
  if (hasExpired) setStatusLED(255, 0, 0);       
  else if (hasWarning) setStatusLED(255, 100, 0); 
  else setStatusLED(0, 255, 0);                  
}

void saveItem(int d, int m) {
  int slot = -1;
  for(int i = 0; i < MAX_ITEMS; i++) if(!inventory[i].active) { slot = i; break; }
  
  if(slot != -1) {
    strncpy(inventory[slot].name, itemNameInput, 8);
    inventory[slot].expiryDay = d; 
    inventory[slot].expiryMonth = m;
    inventory[slot].active = true;
    
    lcd.clear(); lcd.print("Saving Local...");
    delay(500);

    if(WiFi.status() == WL_CONNECTED) {
      lcd.setCursor(0,1); lcd.print("Pushing Cloud...");
      HTTPClient http;
      http.begin(serverName);
      http.addHeader("Content-Type", "application/json");
      // UPDATED YEAR TO 2026
      String payload = "{\"name\":\"" + String(itemNameInput) + "\", \"expiryDate\":\"2026-" + String(m) + "-" + String(d) + "\"}";
      http.POST(payload);
      http.end();
    }
    lcd.clear(); lcd.print("Item Saved!"); 
    beepBuzzer(1); 
    delay(2000); 
    fetchInventory();
  } else {
    lcd.clear(); lcd.print("Pantry Full!"); delay(2000);
  }
  showIdleScreen();
}

bool canMakeRecipe(int rIdx) {
  const char* ing[3]; 
  int numNeeded = 0;
  if (rIdx == 0) { ing[0]="Milk"; ing[1]="Eggs"; ing[2]="Flour"; numNeeded=3; }
  else if (rIdx == 1) { ing[0]="Eggs"; ing[1]="Cheese"; numNeeded=2; }
  else if (rIdx == 2) { ing[0]="Bread"; ing[1]="Cheese"; numNeeded=2; }
  else return false;

  int count = 0;
  for(int i=0; i<numNeeded; i++) {
    for(int j=0; j<MAX_ITEMS; j++) {
      if(inventory[j].active && strcmp(inventory[j].name, ing[i]) == 0) { count++; break; }
    }
  }
  return (count == numNeeded);
}

void viewRecipes() {
  int first = -1;
  for(int i=0; i<NUM_RECIPES; i++) if(canMakeRecipe(i)) { first=i; break; }
  if(first == -1) { lcd.clear(); lcd.print("No Recipes"); lcd.setCursor(0,1); lcd.print("Possible"); delay(2000); showIdleScreen(); return; }

  recipeIndex = first;
  while(true) {
    lcd.clear(); lcd.print("Can Make:");
    lcd.setCursor(0,1); lcd.print(recipes[recipeIndex]);
    char key = customKeypad.waitForKey();
    if(key == 'B') {
      int next = -1;
      for(int i=recipeIndex+1; i<NUM_RECIPES; i++) if(canMakeRecipe(i)) { next=i; break; }
      recipeIndex = (next != -1) ? next : first;
    } else if(key == '*' || key == 'D') break;
  }
  showIdleScreen();
}

void viewInventory() {
  int first = -1;
  for(int i=0; i<MAX_ITEMS; i++) if(inventory[i].active) { first=i; break; }
  if(first == -1) { lcd.clear(); lcd.print("Pantry Empty"); delay(1500); showIdleScreen(); return; }
  inventoryIndex = first;
  while(true) {
    lcd.clear(); lcd.print(inventory[inventoryIndex].name);
    lcd.setCursor(0,1); lcd.print("Exp: "); lcd.print(inventory[inventoryIndex].expiryDay);
    lcd.print("/"); lcd.print(inventory[inventoryIndex].expiryMonth); lcd.print(" D:Del");
    char key = customKeypad.waitForKey();
    if(key == 'B') { 
      int next = -1;
      for(int i=inventoryIndex+1; i<MAX_ITEMS; i++) if(inventory[i].active) { next=i; break; }
      inventoryIndex = (next != -1) ? next : first;
    } else if(key == 'D') { 
      inventory[inventoryIndex].active = false;
      lcd.clear(); lcd.print("Removed"); delay(1500); break;
    } else if(key == '*' || key == 'A') break;
  }
  showIdleScreen();
}

void setup() {
  Serial.begin(115200);
  pinMode(RED_PIN, OUTPUT); pinMode(GREEN_PIN, OUTPUT); pinMode(BLUE_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(RED_PIN, HIGH); digitalWrite(GREEN_PIN, HIGH); digitalWrite(BLUE_PIN, HIGH);

  Wire.begin(21, 22);
  lcd.init(); lcd.backlight();
  lcd.print("Shelf Starting...");

  dht.begin();
  if (!rtc.begin()) { Serial.println("RTC Fail"); }
  if (!rtc.isrunning()) { rtc.adjust(DateTime(F(__DATE__), F(__TIME__))); }
  
  WiFi.begin(ssid, password);
  unsigned long startT = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startT < 10000) { delay(500); }

  if(WiFi.status() == WL_CONNECTED) fetchInventory(); 
  else { lcd.clear(); lcd.print("Offline Mode"); delay(1500); }

  showIdleScreen();
}

void loop() {
  char key = customKeypad.getKey();
  static unsigned long lastCheck = 0;
  
  if (millis() - lastCheck > 3000) {
    manageAndCheckExpiry();
    lastCheck = millis();
    if(currentState == STATE_IDLE) {
      DateTime now = rtc.now();
      lcd.setCursor(0,0);
      lcd.print(now.day()); lcd.print("/"); lcd.print(now.month()); lcd.print(" ");
      lcd.setCursor(9, 0); lcd.print(currentTemp, 1); lcd.print("C ");
    }
  }

  switch(currentState) {
    case STATE_IDLE:
      if(key == '#') {
        scrollIndex = 0; currentState = STATE_SCROLL_MENU;
        lcd.clear(); lcd.print("Select Item:");
        lcd.setCursor(0,1); lcd.print("> "); lcd.print(predefinedItems[scrollIndex]);
      } else if(key == 'B') viewInventory();
      else if(key == 'C') viewRecipes();
      break;

    case STATE_SCROLL_MENU:
      if(key == 'A') {
        scrollIndex = (scrollIndex <= 0) ? NUM_PREDEFINED - 1 : scrollIndex - 1;
        // Refresh display
        lcd.setCursor(0, 1); lcd.print("                "); // Clear line
        lcd.setCursor(0, 1); lcd.print("> "); lcd.print(predefinedItems[scrollIndex]);
      }
      else if(key == 'B') {
        scrollIndex = (scrollIndex + 1) % NUM_PREDEFINED;
        // Refresh display
        lcd.setCursor(0, 1); lcd.print("                "); // Clear line
        lcd.setCursor(0, 1); lcd.print("> "); lcd.print(predefinedItems[scrollIndex]);
      }
      else if(key == '#') {
        if(strcmp(predefinedItems[scrollIndex], "Other") == 0) {
          memset(itemNameInput, 0, sizeof(itemNameInput));
          charIndex = 0; lastKey = ' '; multiTapIndex = 0;
          lcd.clear(); lcd.print("Enter Name:");
          lcd.setCursor(0,1); lcd.cursor(); currentState = STATE_CUSTOM_ENTRY;
        } else {
          strcpy(itemNameInput, predefinedItems[scrollIndex]);
          lcd.clear(); lcd.print("Expiry DDMM:");
          dateIdx = 0; memset(dateInput, 0, sizeof(dateInput)); currentState = STATE_EXPIRY_ENTRY;
        }
      } else if(key == '*') showIdleScreen();
      break;

    case STATE_CUSTOM_ENTRY:
      if(key) {
        const char* keym[] = {"", "1", "ABC2", "DEF3", "GHI4", "JKL5", "MNO6", "PQRS7", "TUV8", "WXYZ9", "*", " 0", "#"};
        if(key == '#') { 
          lcd.clear(); lcd.print("Expiry DDMM:"); 
          dateIdx = 0; memset(dateInput, 0, sizeof(dateInput)); currentState = STATE_EXPIRY_ENTRY; 
        }
        else if(key == '*') { 
          if(charIndex > 0) { charIndex--; itemNameInput[charIndex] = ' '; lcd.setCursor(charIndex,1); lcd.print(" "); lcd.setCursor(charIndex,1); }
          lastKey = ' ';
        }
        else if(key >= '0' && key <= '9') {
          if(key == lastKey && (millis() - lastKeyTime < multiTapTimeout)) multiTapIndex++;
          else { if(lastKey != ' ') charIndex++; if(charIndex >= 8) charIndex = 7; multiTapIndex = 0; }
          int k = (key == '0') ? 11 : (key - '0');
          const char* chars = keym[k];
          if(strlen(chars) > 0) {
            char c = chars[multiTapIndex % strlen(chars)];
            itemNameInput[charIndex] = c;
            lcd.setCursor(charIndex, 1); lcd.print(c); lcd.setCursor(charIndex, 1);
          }
          lastKey = key; lastKeyTime = millis();
        }
      }
      break;

    case STATE_EXPIRY_ENTRY:
      if(key >= '0' && key <= '9') {
        if(dateIdx < 4) {
          dateInput[dateIdx] = key;
          lcd.print(key);
          dateIdx++;
          if(dateIdx == 4) {
            int d = (dateInput[0]-'0')*10 + (dateInput[1]-'0');
            int m = (dateInput[2]-'0')*10 + (dateInput[3]-'0');
            saveItem(d, m);
          }
        }
      } else if (key == '*') showIdleScreen();
      break;
  }
}
